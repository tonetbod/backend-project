const express = require("express");
const { Client } = require("pg");

const app = express();

const PORT = 3000;

// Connect to PostgreSQL
const db = new Client({
    host: "localhost",
    user: "postgres",
    password: "TOM",
    database: "student_db",
    port: 5050
});


// Test PostgreSQL connection
db.connect((err) => {
    if (err) {
        console.log("PostgreSQL connection failed!");
        console.log(err.message);
        return;
    }

    console.log("PostgreSQL connected successfully!");
});


// Home route
app.get("/", (req, res) => {
    res.send("Node.js + Express + PSQL is working!");
});


// Get students
app.get("/students", (req, res) => {

    db.query("SELECT * FROM students", (err, result) => {

        if (err) {
            console.log(err);
            return res.status(500).send("Database error");
        }

        res.json(result.rows);
    });

});

// Start server
app.listen(PORT, () => {
    console.log(`Server running at http://localhost:${PORT}`);
});

C. Create the GET Route

We want: GET /students

to retrieve all students from PostgreSQL.

Add this to app.js:


The code:

// ========================================
// EXPRESS SERVER
// ========================================

const express = require("express");
const pool = require("./db");

const app = express();

// Allow Express to receive JSON data
app.use(express.json());

const PORT = 3000;


// ========================================
// GET ALL STUDENTS
// ========================================

app.get("/students", async (req, res) => {
    try {

        // Get all students from PostgreSQL
        const result = await pool.query(
            "SELECT * FROM students ORDER BY id ASC"
        );

        res.json(result.rows);

    } catch (error) {

        console.error(error);

        res.status(500).json({
            message: "Error retrieving students"
        });
    }
});


// ========================================
// GET STUDENT BY ID
// ========================================

app.get("/students/:id", async (req, res) => {
    try {

        // Get the ID from the URL
        const id = req.params.id;

        // Search PostgreSQL for the student
        const result = await pool.query(
            "SELECT * FROM students WHERE id = $1",
            [id]
        );

        // Check if student was found
        if (result.rows.length === 0) {
            return res.status(404).json({
                message: "Student record not found"
            });
        }

        // Send the student record
        res.json(result.rows[0]);

    } catch (error) {

        console.error(error);

        res.status(500).json({
            message: "Error searching for student"
        });
    }
});


// ========================================
// START SERVER
// ========================================

app.listen(PORT, () => {
    console.log(`Server running at http://localhost:${PORT}`);
});

// ========================================
// INSERT SERVER
// ========================================

const express = require("express");
const pool = require("./db");

const app = express();


// Allow Express to receive JSON
app.use(express.json());

// Serve files from the public folder
app.use(express.static("public"));

const PORT = 4000;


// ========================================
// TEST DATABASE CONNECTION
// ========================================

app.get("/test", async (req, res) => {
    try {

        const result = await pool.query("SELECT NOW()");

        res.json({
            message: "Insert server and PostgreSQL are connected",
            time: result.rows[0]
        });

    } catch (error) {

        console.error("DATABASE ERROR:", error.message);

        res.status(500).json({
            message: "Database connection failed",
            error: error.message
        });
    }
});


// ========================================
// POST STUDENT
// ========================================

app.post("/students", async (req, res) => {
    try {

        // Get student data from the form
        const { name, age, course, email } = req.body;

        console.log("Student received:", req.body);

        // Insert student into PostgreSQL
        const result = await pool.query(
            `INSERT INTO students (name, age, course, email)
             VALUES ($1, $2, $3, $4)
             RETURNING *`,
            [name, age, course, email]
        );

        console.log("Student inserted:", result.rows[0]);

        res.status(201).json({
            message: "Student record added successfully",
            student: result.rows[0]
        });

    } catch (error) {

        console.error("INSERT ERROR:", error.message);

        res.status(500).json({
            message: "Student was not inserted",
            error: error.message
        });
    }
});


// ========================================
// START SERVER
// ========================================

app.listen(PORT, () => {
    console.log(`Insert server running at http://localhost:${PORT}`);
});

    <script>

        const form = document.getElementById("studentForm");

        form.addEventListener("submit", async function(event) {

            // Stop the page from refreshing
            event.preventDefault();


            // Get the values from the form
            const student = {

                name: document.getElementById("name").value,

                age: document.getElementById("age").value,

                course: document.getElementById("course").value,

                email: document.getElementById("email").value

            };


            // Send the student to Express
            const response = await fetch("/students", {

                method: "POST",

                headers: {
                    "Content-Type": "application/json"
                },

                body: JSON.stringify(student)

            });


            // Get the response from Express
            const data = await response.json();


            // Display the result
            document.getElementById("message").textContent =
                data.message;


            // Clear the form if successful
            if (response.ok) {
                form.reset();
            }

        });

    </script>

</body>

</html>
